function blankSentence(v) {
  if (v.word.includes(' ')) return null;
  const stem = v.word.length > 4 && /[ey]$/.test(v.word) ? v.word.slice(0, -1) : v.word;
  const re = new RegExp('\\b' + stem.replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + '\\w*', 'i');
  return re.test(v.sentence) ? v.sentence.replace(re, '＿＿＿＿＿') : null;
}
function distract(v, key, n = 3) {
  const same = vocabData.filter(x => x.id !== v.id && x.pos === v.pos && x[key] !== v[key]);
  const pool = same.length >= n ? same : vocabData.filter(x => x.id !== v.id && x[key] !== v[key]);
  return shuffle(pool).slice(0, n);
}
function makeQ(v, mode) {
  if (mode === 'mix') mode = ['sent', 'e2c', 'c2e', 'spell'][Math.random() * 4 | 0];
  if (mode === 'sent' && !blankSentence(v)) mode = 'e2c';
  const opts = key => shuffle([v, ...distract(v, key)]);
  if (mode === 'sent') return { v, mode, text: blankSentence(v), sub: v.meaning, opts: opts('word').map(x => ({ t: x.word, ok: x.id === v.id })) };
  if (mode === 'e2c') return { v, mode, text: `<b>${esc(v.word)}</b> <span class="pos">${esc(v.pos)}</span>`, html: 1, opts: opts('meaning').map(x => ({ t: x.meaning, ok: x.id === v.id })) };
  if (mode === 'c2e') return { v, mode, text: `${esc(v.meaning)} <span class="pos">${esc(v.pos)}</span>`, html: 1, opts: opts('word').map(x => ({ t: x.word, ok: x.id === v.id })) };
  return { v, mode: 'spell', text: `${esc(v.meaning)} <span class="pos">${esc(v.pos)}</span> <span class="count">（${v.word.replace(/[^ ]/g, '_').length} 字元，首字母 ${esc(v.word[0])}）</span>`, html: 1 };
}

    const [chk, rst] = d.querySelectorAll('button');
    chk.onclick = () => { let s = 0; for (let i = 1; i <= 10; i++) { const el = $(`#p${pi}-${i}`), ok = el.value === p.answers[i - 1]; el.classList.remove('ok', 'bad'); el.classList.add(ok ? 'ok' : 'bad'); if (ok) s += 10; }
      $(`#r${pi}`).innerHTML = `得分：<span style="color:${s >= 60 ? 'var(--ok)' : 'var(--no)'}">${s} / 100</span>`; $(`#e${pi}`).style.display = 'block'; };
    rst.onclick = () => { for (let i = 1; i <= 10; i++) { const el = $(`#p${pi}-${i}`); el.value = ''; el.classList.remove('ok', 'bad'); } $(`#r${pi}`).textContent = ''; $(`#e${pi}`).style.display = 'none'; };
